using Blazored.LocalStorage;
using Microsoft.AspNetCore.Components.Authorization;
using MudBlazor;
using System.Security.Claims;
using System.Text.Json;

namespace OIL.Shared.Services;

public class CustomAuthStateProvider : AuthenticationStateProvider
{
    private readonly ILocalStorageService _localStorage;

    private readonly ClaimsPrincipal _anonymous =
        new(new ClaimsIdentity());

    public CustomAuthStateProvider(
        ILocalStorageService localStorage)
    {
        _localStorage = localStorage;
    }


    public async Task AdminUpdateAuthenticationStateAsync(string email, string role)
    {
        try
        {
            // Secure the token state on MAUI local storage so re-renders remember the context
            var session = new UserSession
            {
                Email = email,
                Role = role
            };

            await _localStorage.SetItemAsync("user_session", JsonSerializer.Serialize(session));

            var claims = new List<Claim>
    {
        new Claim(ClaimTypes.Name, email),
        new Claim(ClaimTypes.Role, role)
    };

            var identity = new ClaimsIdentity(claims, "CustomAuth");
            var user = new ClaimsPrincipal(identity);
            var authState = Task.FromResult(new AuthenticationState(user));

            NotifyAuthenticationStateChanged(authState);
        }
        catch (Exception ex)
        {
            Console.WriteLine($"AdminUpdateAuthenticationStateAsync Error: {ex.Message}");
            //Snackbar.Add($"Error : {ex.Message}", Severity.Error);
        }



    }

    public override async Task<AuthenticationState>
        GetAuthenticationStateAsync()
    {
        try
        {
            var userJson =
                await _localStorage.GetItemAsync<string>("user_session");

            if (string.IsNullOrWhiteSpace(userJson))
                return new AuthenticationState(_anonymous);

            var user =
                JsonSerializer.Deserialize<UserSession>(userJson);

            if (user == null)
                return new AuthenticationState(_anonymous);

            return new AuthenticationState(
                CreateClaimsPrincipal(user.Email, user.Role));
        }
        catch (Exception ex)
        {
            Console.WriteLine($"GetAuthenticationStateAsync Error: {ex.Message}");
            return new AuthenticationState(_anonymous);
        }
    }

    public async Task UpdateAuthenticationState(
        string email,
        string role)
    {

        try
        {
            var session = new UserSession
            {
                Email = email,
                Role = role
            };

            await _localStorage.SetItemAsync(
                "user_session",
                JsonSerializer.Serialize(session));

            var principal =
                CreateClaimsPrincipal(email, role);

            NotifyAuthenticationStateChanged(
                Task.FromResult(
                    new AuthenticationState(principal)));
        }
        catch (Exception ex)
        {
            Console.WriteLine($"UpdateAuthenticationState Error: {ex.Message}");
            //Snackbar.Add($"Error : {ex.Message}", Severity.Error);
        }



    }

    public async Task NotifyLogout()
    {
        try
        {
            await _localStorage.RemoveItemAsync("user_session");

            NotifyAuthenticationStateChanged(
                Task.FromResult(
                    new AuthenticationState(_anonymous)));
        }
        catch (Exception ex)
        {
            Console.WriteLine($"NotifyLogout Error: {ex.Message}");
            //Snackbar.Add($"Error : {ex.Message}", Severity.Error);
        }


    }

    private ClaimsPrincipal CreateClaimsPrincipal(
        string email,
        string role)
    {
        var identity = new ClaimsIdentity(
        [
            new Claim(ClaimTypes.Name,email),
            new Claim(ClaimTypes.Role,role)
        ],
        "CustomAuth");

        return new ClaimsPrincipal(identity);
    }
}

public class UserSession
{
    public string Email { get; set; }
    public string Role { get; set; }
}
