@code {
    [Table("locations")]
    public class Location : BaseModel
    {
        // Second parameter set to false skips the field on DB updates and inserts
        [PrimaryKey("id", false)]
        public long Id { get; set; }

        [Column("sub_area")] public string? SubArea { get; set; }
        [Column("area_name")] public string? AreaName { get; set; }
        [Column("installation_name")] public string? InstallationName { get; set; }
        [Column("field_type")] public string? FieldType { get; set; }
        [Column("image_url")] public string? ImageUrl { get; set; }
        [Column("installation_type")] public string? InstallationType { get; set; }
        [Column("is_active")] public bool? IsActive { get; set; } = true;
        [Column("distance_fhq_km")] public float? DistanceFhqKm { get; set; }
    }

    private List<Location> _locations = new();
    private Location _editingLoc = new();
    private bool _loading = true;
    private bool _isModalOpen = false;

    protected override async Task OnInitializedAsync() => await LoadData();

    private async Task LoadData()
    {
        try
        {
            _loading = true;
            var result = await SupabaseClient.From<Location>()
                .Order(x => x.Id, Supabase.Postgrest.Constants.Ordering.Ascending)
                .Get();
            _locations = result.Models;
            StateHasChanged();
        }
        catch (Exception ex)
        {
            Snackbar.Add($"Load Error: {ex.Message}", Severity.Error);
        }
        finally
        {
            _loading = false;
        }
    }

    private void OpenAddModal()
    {
        _editingLoc = new Location { IsActive = true }; // Instantiates with Id = 0
        _isModalOpen = true;
    }

    private void OpenEditModal(Location l)
    {
        _editingLoc = new Location
        {
            Id = l.Id,
            SubArea = l.SubArea,
            AreaName = l.AreaName,
            InstallationName = l.InstallationName,
            FieldType = l.FieldType,
            ImageUrl = l.ImageUrl,
            InstallationType = l.InstallationType,
            IsActive = l.IsActive,
            DistanceFhqKm = l.DistanceFhqKm
        };
        _isModalOpen = true;
    }

    private async Task SaveLoc()
    {
        if (string.IsNullOrWhiteSpace(_editingLoc.InstallationName))
        {
            Snackbar.Add("Installation Name is required.", Severity.Warning);
            return;
        }

        try
        {
            if (_editingLoc.Id == 0)
            {
                // The database handles row assignment safely using your updated identity configuration
                await SupabaseClient.From<Location>().Insert(_editingLoc);
                Snackbar.Add("Location added successfully", Severity.Success);
            }
            else
            {
                await SupabaseClient.From<Location>().Update(_editingLoc);
                Snackbar.Add("Location data updated", Severity.Info);
            }

            _isModalOpen = false;
            await LoadData();
        }
        catch (Exception ex)
        {
            Snackbar.Add($"Save Failed: {ex.Message}", Severity.Error);
        }
    }

    private async Task ConfirmDelete(Location l)
    {
        var result = await DialogService.ShowMessageBoxAsync(
            "Confirm Removal",
            $"Delete location: {l.InstallationName}?",
            yesText: "Delete",
            cancelText: "Cancel"
        );

        if (result == true)
        {
            try
            {
                await SupabaseClient.From<Location>().Where(x => x.Id == l.Id).Delete();
                await LoadData();
                Snackbar.Add("Location removed", Severity.Warning);
            }
            catch (Exception ex)
            {
                Snackbar.Add($"Delete Failed: {ex.Message}", Severity.Error);
            }
        }
    }
}
