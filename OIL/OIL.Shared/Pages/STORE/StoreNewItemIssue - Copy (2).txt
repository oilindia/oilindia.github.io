    private async Task LoadWorkspaceDataAsync()
    {
        try
        {
            isLoading = true;
            errorMessage = string.Empty;

            // 1. Calculate the 30-day cutoff timestamp (ISO-8601 string)
            string cutoffDate = DateTime.UtcNow.AddDays(-30).ToString("o");

            // 2. Fetch jobs sorted with latest records at the top (last 30 days only)
            var jobsResponse = await Supabase.From<Job>()
                .Filter("created_at", global::Supabase.Postgrest.Constants.Operator.GreaterThanOrEqual, cutoffDate)
                .Order("created_at", global::Supabase.Postgrest.Constants.Ordering.Descending)
                .Get();
            jobsList = jobsResponse.Models;

            // 3. Fetch job assignments sorted with latest records at the top (last 30 days only)
            var assignmentsResponse = await Supabase.From<JobAssignment>()
                .Filter("created_at", global::Supabase.Postgrest.Constants.Operator.GreaterThanOrEqual, cutoffDate)
                .Order("created_at", global::Supabase.Postgrest.Constants.Ordering.Descending)
                .Get();
            assignmentsList = assignmentsResponse.Models;

            // 4. Fetch your engineers directory for array resolution mapping (Keep unfiltered as it's a structural lookup)
            var engineersResponse = await Supabase.From<EmployeeEngineer>().Get();
            engineersDirectory = engineersResponse.Models;


        }
        catch (Exception ex)
        {
            errorMessage = $"Error populating workbench cache tables: {ex.Message}";
        }
        finally
        {
            isLoading = false;
            StateHasChanged();
        }
    }

    private async Task SaveDepartmentAllocationsAsync()
    {
        if (!departmentIssuesList.Any()) return;

        // 1. Structural Checks
        if (departmentIssuesList.Any(x => x.IssuedQuantity <= 0))
        {
            deptErrorMessage = "Allocation Execution Failed: Issued Quantity must be greater than 0.";
            return;
        }

        // 2. Strict Inventory Level Balance Checks
        if (departmentIssuesList.Any(x => x.IssuedQuantity > x.SapStock))
        {
            deptErrorMessage = "Allocation Execution Failed: Insufficient warehouse stock level found on structural rows.";
            return;
        }

        deptErrorMessage = string.Empty;
        isLoading = true;

        try
        {
            await Task.Delay(600); // Simulate network latency block

            // Loop and automatically reduce stock quantities inside master memory arrays
            foreach (var record in departmentIssuesList)
            {
                var masterItem = allSpares.FirstOrDefault(s => s.Id == record.MaterialId);
                if (masterItem != null)
                {
                    // Automatic transactional reduction
                    masterItem.StockQuantity -= (decimal)record.IssuedQuantity;
                }
            }

            // Flush the workspace batch on successful processing transaction run
            departmentIssuesList.Clear();
        }
        catch (Exception ex)
        {
            deptErrorMessage = $"Database Sync Exception: {ex.Message}";
        }
        finally
        {
            isLoading = false;
        }
    }

    private async Task ConfirmAndCommitBatch()
    {
        if (selectedJob == null || !stagedSparesBucket.Any()) return;

        // 1. Guard check: Validate input states before hitting production tables
        foreach (var item in stagedSparesBucket)
        {
            if (item.QuantityToIssue <= 0 || item.QuantityToIssue > item.StockQuantity)
            {
                errorMessage = $"Stock boundaries violated for code: {item.Material}. Check entry limits.";
                return;
            }
        }
        //StateHasChanged();
        try
        {
            isLoading = true; // Lock down components during async context calls
            errorMessage = string.Empty;

            // Find matching assignment ID from your existing UI memory dataset if available
            var matchedAssignment = assignmentsList.FirstOrDefault(a => a.JobIds != null && a.JobIds.Contains(selectedJob.Id));

            // 2. STEP ONE: Insert Parent Record (The Transaction Header)
            var issuanceHeader = new MaterialIssuance
            {

    public class JobIssuedSpareSummary
    {
        public long MaterialId { get; set; }
        public string MaterialCode { get; set; } = string.Empty;
        public string Description { get; set; } = string.Empty;
        public decimal TotalIssued { get; set; }
        public decimal TotalReturned { get; set; }
        public decimal NetAllocated => TotalIssued - TotalReturned;
        public decimal UnitCost { get; set; }
        public decimal QuantityToReturnInput { get; set; } // Tracks real-time UI text box entry
    }

    private List<JobIssuedSpareSummary> jobIssuedSparesSummaryList = new();

    private async Task LoadJobSparesHistoryAsync(Guid jobId)
    {
        try
        {
            jobIssuedSparesSummaryList.Clear();
            errorMessage = string.Empty;

            // 1. Fetch all issuance IDs for this job ticket
            var issuanceHeaders = await Supabase.From<MaterialIssuance>()
                .Where(x => x.JobId == jobId)
                .Get();
            var issuanceIds = issuanceHeaders.Models.Select(x => x.Id).ToList();

            List<MaterialIssuanceItem> issuedItems = new();
            if (issuanceIds.Any())
            {
                var itemsResponse = await Supabase.From<MaterialIssuanceItem>()
                    .Filter("issuance_id", global::Supabase.Postgrest.Constants.Operator.In, issuanceIds)
                    .Get();
                issuedItems = itemsResponse.Models;
            }

            // 2. Fetch all return records for this job ticket
            var returnHeaders = await Supabase.From<MaterialReturn>()
                .Where(x => x.JobId == jobId)
                .Get();
            var returnIds = returnHeaders.Models.Select(x => x.Id).ToList();

            List<MaterialReturnItem> returnedItems = new();
            if (returnIds.Any())
            {
                var returnsResponse = await Supabase.From<MaterialReturnItem>()
                    .Filter("return_id", global::Supabase.Postgrest.Constants.Operator.In, returnIds)
                    .Get();
                returnedItems = returnsResponse.Models;
            }

            // 3. Load descriptions for all involved materials
            var uniqueMaterialIds = issuedItems.Select(x => x.MaterialId).Distinct().ToList();
            List<StoreMaterial> materialsLookup = new();
            if (uniqueMaterialIds.Any())
            {
                var matsResponse = await Supabase.From<StoreMaterial>()
                    .Filter("id", global::Supabase.Postgrest.Constants.Operator.In, uniqueMaterialIds.Cast<object>().ToList())
                    .Get();
                materialsLookup = matsResponse.Models;
            }

            // 4. Build the merged view model matrix
            foreach (var matId in uniqueMaterialIds)
            {
                var matInfo = materialsLookup.FirstOrDefault(x => x.Id == matId);
                var totalIssued = issuedItems.Where(x => x.MaterialId == matId).Sum(x => x.QuantityIssued);
                var totalReturned = returnedItems.Where(x => x.MaterialId == matId).Sum(x => x.QuantityReturned);
                var pricing = issuedItems.Where(x => x.MaterialId == matId).Select(x => x.UnitCostAtIssue).FirstOrDefault();

                jobIssuedSparesSummaryList.Add(new JobIssuedSpareSummary
                {
                    MaterialId = matId,
                    MaterialCode = matInfo?.Material ?? $"ID: {matId}",
                    Description = matInfo?.MaterialDescription ?? "Unknown Part Profile",
                    TotalIssued = totalIssued,
                    TotalReturned = totalReturned,
                    UnitCost = pricing
                });
            }
        }
        catch (Exception ex)
        {
            errorMessage = $"Error computing historical logs: {ex.Message}";
        }
    }

    // Auto-Clear Error Boundary Code
    private void OnIssueQuantityChanged(StagedSpareItem item, string value)
    {
        if (decimal.TryParse(value, out var parsedQty))
        {
            item.QuantityToIssue = (int)parsedQty;
        }
        else
        {
            item.QuantityToIssue = 0;
        }

        // Evaluate current list boundaries
        bool boundariesViolated = false;
        foreach (var staged in stagedSparesBucket)
        {
            if (staged.QuantityToIssue <= 0 || staged.QuantityToIssue > staged.StockQuantity)
            {
                boundariesViolated = true;
                errorMessage = $"Stock boundaries violated for code: {staged.Material}. Check entry limits.";
                break;
            }
        }

        // Clear error message if all items are valid
        if (!boundariesViolated)
        {
            errorMessage = string.Empty;
        }
    }

    // Process returns directly from the loaded grid row items
    private async Task CommitGridReturnsAsync()
    {
        var activeReturns = jobIssuedSparesSummaryList.Where(x => x.QuantityToReturnInput > 0).ToList();
        if (selectedJob == null || !activeReturns.Any()) return;

        foreach (var item in activeReturns)
        {
            if (item.QuantityToReturnInput > item.NetAllocated)
            {
                errorMessage = $"Cannot return more than net allocated for code: {item.MaterialCode}";
                return;
            }
        }

        try
        {
            isLoading = true;
            errorMessage = string.Empty;

            var returnHeader = new MaterialReturn
            {
                JobId = selectedJob.Id,
                ReturnedByEmail = "surendra.choudhary@company.com",
                Remarks = "Direct store-counter refund processing"
            };

            var headerResponse = await Supabase.From<MaterialReturn>().Insert(returnHeader);
            var committedHeader = headerResponse.Model;

            var itemsToReturn = activeReturns.Select(item => new MaterialReturnItem
            {
                ReturnId = committedHeader.Id,
                MaterialId = item.MaterialId,
                QuantityReturned = item.QuantityToReturnInput,
                UnitCostAtReturn = item.UnitCost
            }).ToList();

            await Supabase.From<MaterialReturnItem>().Insert(itemsToReturn);

            // Refresh UI state values
            await LoadWorkspaceDataAsync();
            await LoadJobSparesHistoryAsync(selectedJob.Id);
            selectedJob = jobsList.FirstOrDefault(j => j.Id == selectedJob.Id);
        }
        catch (Exception ex)
        {
            errorMessage = $"Return execution failure: {ex.Message}";
        }
        finally
        {
            isLoading = false;
            StateHasChanged();
        }
    }

    // Staging Mechanism: Finds the item based on the typed query and moves it to the returns bucket
    private async Task StageItemByCode()
    {
        if (string.IsNullOrWhiteSpace(returnSearchInput)) return;

        returnUiFeedbackMessage = string.Empty;

        try
        {
            // 1. Query the master database index to confirm this material exists
            var lookupResponse = await Supabase.From<StoreMaterial>()
                .Where(x => x.Material == returnSearchInput.Trim())
                .Limit(1)
                .Get();

            var targetedMaterial = lookupResponse.Models.FirstOrDefault();

            if (targetedMaterial == null)
            {
                returnUiFeedbackMessage = "Material item code not found in warehouse master databases.";
                return;
            }

            // 2. Guard against duplicating items within the staging bucket list
            if (stagedReturnsBucket.Any(x => x.MaterialId == targetedMaterial.Id))
            {
                returnUiFeedbackMessage = "This material item is already staged in the manifest list below.";
                return;
            }

            // 3. Append safely into the return model bucket array
            stagedReturnsBucket.Add(new MaterialReturnStagingModel
            {
                MaterialId = targetedMaterial.Id,
                MaterialCode = targetedMaterial.Material,
                Description = targetedMaterial.MaterialDescription,
                QuantityToReturn = 1, // Default fallback start value
                UnitCost = targetedMaterial.UnitCost
            });

            // Clear input box for next entry scan
            returnSearchInput = string.Empty;
        }
        catch (Exception ex)
        {
            returnUiFeedbackMessage = $"Search resolution failure: {ex.Message}";
        }
    }

    // Removal Mechanism: Clears an individual row out of the staging layout list
    private void RemoveFromReturnStaging(MaterialReturnStagingModel item)
    {
        stagedReturnsBucket.Remove(item);
    }

    private async Task CommitMaterialReturnBatch()
    {
        if (selectedJob == null || !stagedReturnsBucket.Any()) return;

        try
        {
            isLoading = true;
            errorMessage = string.Empty;

            // 1. Write the Return Transaction Header Record
            var returnHeader = new MaterialReturn
            {
                JobId = selectedJob.Id,
                ReturnedByEmail = "surendra.choudhary@company.com", // Bind to auth context state
                Remarks = $"Unused material returned to store warehouse storage lines for Job: {selectedJob.JobName}"
            };

            var headerResponse = await Supabase.From<MaterialReturn>().Insert(returnHeader);
            var committedHeader = headerResponse.Model;

            if (committedHeader == null || committedHeader.Id == Guid.Empty)
            {
                throw new Exception("Database failed to initialize the Return Tracking ID.");
            }

            // 2. Map the UI Staging bucket array to database entity models
            var itemsToReturn = new List<MaterialReturnItem>();
            foreach (var item in stagedReturnsBucket)
            {
                itemsToReturn.Add(new MaterialReturnItem
                {
                    ReturnId = committedHeader.Id,
                    MaterialId = item.MaterialId,
                    QuantityReturned = item.QuantityToReturn,
                    UnitCostAtReturn = item.UnitCost
                });
            }

            // 3. Fire the array batch insert to Supabase
            await Supabase.From<MaterialReturnItem>().Insert(itemsToReturn);

            // 4. Wipe operational staging states clean
            stagedReturnsBucket.Clear();

            // 5. Re-sync workspace datasets to display the restored stock values instantly
            await LoadWorkspaceDataAsync();
            if (selectedJob != null)
            {
                selectedJob = jobsList.FirstOrDefault(j => j.Id == selectedJob.Id);
            }
        }
        catch (Exception ex)
        {
            errorMessage = $"Stock restoration breakdown: {ex.Message}";
        }
        finally
        {
            isLoading = false;
            StateHasChanged();
        }
    }
