    private bool _loading = false;

    private DateTime? _startDate = DateTime.Today.AddMonths(-1);

    private DateTime? _endDate = DateTime.Today;

   

    // Constants for Cost calculations
    private const double HourlyOvertimeRate = 750.00;
    private const double StandardEmployeeDayRate = 2500.00;
    private const double FixedTravelCostPerAssignment = 1200.00;

    // Metrics Data Aggregates
    public double TotalCalculatedCost { get; set; }
    public double TotalOvertimeHours { get; set; }
    public int TotalPmReportsMatched { get; set; }
    public int TotalAssignmentsProcessed { get; set; }

    // Proportional breakdown metrics
    public double TotalOvertimeCost { get; set; }
    public double TotalSparesCost { get; set; }

    public double TotalTravelCost { get; set; }

    private async Task RefreshDashboardAsync()
    {
        _loading = true;
        StateHasChanged();

        // Clear previous state records explicitly
        TotalCalculatedCost = 0; TotalOvertimeHours = 0; TotalPmReportsMatched = 0; TotalAssignmentsProcessed = 0;
        TotalOvertimeCost = 0; TotalSparesCost = 0; TotalEmployeeCost = 0; TotalTravelCost = 0;
        GridRows.Clear();
        DailyCostSeries.Clear();

        List<JobAssignmentModel> assignments = new();
        List<PmReportModel> pmReports = new();
        List<JobModel> jobs = new();

        _loading = true;

        var start = (_startDate ?? DateTime.Today.AddMonths(-1)).Date;
        var end = (_endDate ?? DateTime.Today).Date.AddDays(1);

        // 1. ISOLATED CONCURRENT DATABASE FETCH LAYERS WITH EXPLICIT SNACKBAR TRACING
        try
        {
           

            var assignmentResponse = await SupabaseClient.From<JobAssignmentModel>()
            .Select("*")
                           .Filter("created_at", Operator.GreaterThanOrEqual, start.ToString("yyyy-MM-dd"))
                           .Filter("created_at", Operator.LessThanOrEqual, end.ToString("yyyy-MM-dd"))
            .Get();

            assignments = assignmentResponse.Models ?? new List<JobAssignmentModel>();

            Snackbar.Add($"Loaded {assignments.Count} assignments from Supabase.", Severity.Info);
        }
        catch (Exception ex)
        {
            Snackbar.Add($"Database Fetch Error (job_assignments table): {ex.Message}", Severity.Error);
        }

        try
        {
            var pmReportResponse = await SupabaseClient.From<PmReportModel>()
            .Select("*")
                           .Filter("completion_date", Operator.GreaterThanOrEqual, start.ToString("yyyy-MM-dd"))
                           .Filter("completion_date", Operator.LessThanOrEqual, end.ToString("yyyy-MM-dd"))
            .Get();
            pmReports = pmReportResponse.Models ?? new List<PmReportModel>();
            Snackbar.Add($"Loaded {pmReports.Count} production PM reports from Supabase.", Severity.Info);
        }
        catch (Exception ex)
        {
            Snackbar.Add($"Database Fetch Error (pm_reports table): {ex.Message}", Severity.Error);
        }

        try
        {
            var jobsResponse = await SupabaseClient.From<JobModel>()
            .Select("*")
                           .Filter("completion_date", Operator.GreaterThanOrEqual, start.ToString("yyyy-MM-dd"))
                           .Filter("completion_date", Operator.LessThanOrEqual, end.ToString("yyyy-MM-dd"))
            .Get();
            jobs = jobsResponse.Models ?? new List<JobModel>();
            Snackbar.Add($"Loaded {jobs.Count} raw maintenance jobs from Supabase.", Severity.Info);
        }
        catch (Exception ex)
        {
            Snackbar.Add($"Database Fetch Error (jobs table): {ex.Message}", Severity.Error);
        }

        // 2. DATA PIPELINE FILTERS & AGGREGATIONS
        try
        {
            if (_startDate.HasValue)
            {
                assignments = assignments.Where(a => a.CreatedAt >= _startDate.Value).ToList();
            }
            if (_endDate.HasValue)
            {
                assignments = assignments.Where(a => a.CreatedAt <= _endDate.Value).ToList();
            }

            TotalAssignmentsProcessed = assignments.Count;

            // Pre-calculate distinct chronological array boundaries for daily grouping pipeline
            var uniqueDates = assignments
                .Select(a => a.CreatedAt.Date)
                .Distinct()
                .OrderBy(d => d)
                .ToList();

            DailyLabels = uniqueDates.Select(d => d.ToString("dd MMM")).ToArray();

            // Setup temporary tracking arrays matching the size of found unique dates
            double[] dailyOvertime = new double[uniqueDates.Count];
            double[] dailySpares = new double[uniqueDates.Count];
            double[] dailyEmployee = new double[uniqueDates.Count];
            double[] dailyTravel = new double[uniqueDates.Count];

            var dateToIndexMap = uniqueDates
                .Select((date, index) => new { date, index })
                .ToDictionary(x => x.date, x => x.index);

            double[] overtimeCosts = new double[4];
            double[] sparesCosts = new double[4];
            double[] employeeCosts = new double[4];
            double[] travelCosts = new double[4];

            foreach (var assignment in assignments)
            {
                if (assignment.CreatedAt == DateTime.MinValue) continue;

                int quarter = (assignment.CreatedAt.Month - 1) / 3;
                if (quarter < 0 || quarter > 3) quarter = 0;

                // Create row record item for data grid tracking ledger
                var gridRowItem = new AssignmentGridRow
                {
                    AssignmentId = assignment.Id,
                    CreatedAt = assignment.CreatedAt
                };

                // A. Overtime Spend Parsing Engine
                double otHours = assignment.JobOvertime ?? 0;
                TotalOvertimeHours += otHours;
                double calculatedOtCost = otHours * HourlyOvertimeRate;
                overtimeCosts[quarter] += calculatedOtCost;
                TotalOvertimeCost += calculatedOtCost;

                gridRowItem.OvertimeHours = otHours;
                gridRowItem.OvertimeCost = calculatedOtCost;

                // B. Employee Resource Parsing Engine
                int personnelCount = assignment.PersonnelIds?.Count ?? 1;
                double calculatedEmployeeCost = personnelCount * StandardEmployeeDayRate;
                employeeCosts[quarter] += calculatedEmployeeCost;
                TotalEmployeeCost += calculatedEmployeeCost;

                gridRowItem.PersonnelCount = personnelCount;
                gridRowItem.EmployeeCost = calculatedEmployeeCost;

                // C. Logistics/Travel Spend Mapping Engine
                double calculatedTravelCost = FixedTravelCostPerAssignment;
                travelCosts[quarter] += calculatedTravelCost;
                TotalTravelCost += calculatedTravelCost;

                gridRowItem.TravelCost = calculatedTravelCost;

                // D. Spares Component Dynamic Deep Join Mapping Engine
                double itemSparesCost = 0;
                if (assignment.JobIds != null && assignment.JobIds.Any())
                {
                    foreach (var jobId in assignment.JobIds)
                    {
                        if (string.IsNullOrWhiteSpace(jobId)) continue;

                        var matchedPm = pmReports.FirstOrDefault(p => string.Equals(p.JobId, jobId, StringComparison.OrdinalIgnoreCase));
                        if (matchedPm != null)
                        {
                            itemSparesCost += matchedPm.ApproxTotalCost ?? 0;
                            TotalPmReportsMatched++;
                        }
                        else
                        {
                            var fallbackJob = jobs.FirstOrDefault(j => string.Equals(j.Id, jobId, StringComparison.OrdinalIgnoreCase));
                            if (fallbackJob != null)
                            {
                                itemSparesCost += fallbackJob.PartsCost ?? 0;
                            }
                        }
                    }
                }
                sparesCosts[quarter] += itemSparesCost;
                TotalSparesCost += itemSparesCost;
                gridRowItem.SparesCost = itemSparesCost;

                // Row total calculation step
                gridRowItem.RowTotalCost = calculatedOtCost + calculatedEmployeeCost + calculatedTravelCost + itemSparesCost;
                GridRows.Add(gridRowItem);

                // Populate Daily Accumulator Buckets
                if (dateToIndexMap.TryGetValue(assignment.CreatedAt.Date, out int dayIndex))
                {
                    dailyOvertime[dayIndex] += calculatedOtCost;
                    dailyEmployee[dayIndex] += calculatedEmployeeCost;
                    dailyTravel[dayIndex] += calculatedTravelCost;
                    dailySpares[dayIndex] += itemSparesCost;
                }
            }

            // 3. COMPILE CHART COMPONENT DATA SERIES ARRAYS
            QuarterlyCostSeries = new List<ChartSeries<double>>
            {
                new ChartSeries<double> { Name = "Overtime Cost", Data = overtimeCosts },
                new ChartSeries<double> { Name = "Spares Cost", Data = sparesCosts },
                new ChartSeries<double> { Name = "Employee Cost", Data = employeeCosts },
                new ChartSeries<double> { Name = "Travel Cost", Data = travelCosts }
            };

            // Assemble the Daily Stacked Bar Series Data
            DailyCostSeries = new List<ChartSeries<double>>
            {
                new ChartSeries<double> { Name = "Overtime Cost", Data = dailyOvertime },
                new ChartSeries<double> { Name = "Spares Cost", Data = dailySpares },
                new ChartSeries<double> { Name = "Employee Cost", Data = dailyEmployee },
                new ChartSeries<double> { Name = "Travel Cost", Data = dailyTravel }
            };

            // Set global absolute totals
            TotalCalculatedCost = TotalOvertimeCost + TotalSparesCost + TotalEmployeeCost + TotalTravelCost;

            // Map proportional share composition arrays
            ProportionalShareData = new double[] { TotalOvertimeCost, TotalSparesCost, TotalEmployeeCost, TotalTravelCost };
        }
        catch (Exception pipelineEx)
        {
            Snackbar.Add($"Data Processing Loop Fault: {pipelineEx.Message}", Severity.Error);
        }
        finally
        {
            _loading = false;
            StateHasChanged();
        }
    }
